## 2. Skema DDL Database Supabase (PostgreSQL)

Jalankan perintah SQL berikut pada menu **SQL Editor** di dashboard Supabase (`https://supabase.com/dashboard/project/jaizwaoqbqykvyvdxjyj/sql`).

Script ini membuat **2 tabel**:
1. `kriteria` : Menyimpan parameter 8 kriteria perhitungan ROC & TOPSIS.
2. `laptops` : Tabel kosong untuk menampung input manual data laptop (Ready vs Indent).

```sql
-- ==============================================================================
-- 1. TABEL MASTER KRITERIA (8 Kriteria Analisis Multi-Kriteria)
-- ==============================================================================
CREATE TABLE IF NOT EXISTS public.kriteria (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    kode VARCHAR(10) UNIQUE NOT NULL,
    nama VARCHAR(100) NOT NULL,
    tipe VARCHAR(10) CHECK (tipe IN ('Benefit', 'Cost')) NOT NULL,
    satuan VARCHAR(50) NOT NULL,
    keterangan TEXT,
    default_rank INT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- ==============================================================================
-- 2. TABEL DATA LAPTOP (STATUS READY TOKO VS INDENT) - TANPA DATA CONTOH
-- ==============================================================================
CREATE TABLE IF NOT EXISTS public.laptops (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nama VARCHAR(150) NOT NULL,
    merek VARCHAR(50) NOT NULL,
    status VARCHAR(20) CHECK (status IN ('ready', 'indent')) NOT NULL DEFAULT 'ready',
    kategori_penggunaan VARCHAR(50) DEFAULT 'Umum',
    
    -- Parameter Nilai Kriteria untuk Komputasi TOPSIS
    harga NUMERIC(15, 2) NOT NULL,      -- C1 (Cost)
    cpu_score NUMERIC(6, 2) NOT NULL,   -- C2 (Benefit, Skala 0-100)
    ram_gb INT NOT NULL,                -- C3 (Benefit, GB)
    ssd_gb INT NOT NULL,                -- C4 (Benefit, GB)
    gpu_score NUMERIC(6, 2) NOT NULL,   -- C5 (Benefit, Skala 0-100)
    baterai_wh NUMERIC(6, 2) NOT NULL,  -- C6 (Benefit, Wh)
    berat_kg NUMERIC(4, 2) NOT NULL,    -- C7 (Cost, Kg)
    layar_score NUMERIC(6, 2) NOT NULL, -- C8 (Benefit, Skala 0-100)

    spesifikasi_ringkas TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- ==============================================================================
-- 3. PENGATURAN KEAMANAN & AKSES (Row Level Security)
-- ==============================================================================
ALTER TABLE public.kriteria ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.laptops ENABLE ROW LEVEL SECURITY;

-- Kebijakan Akses Baca (SELECT) untuk Publik
CREATE POLICY "Public Read Kriteria" ON public.kriteria FOR SELECT USING (true);
CREATE POLICY "Public Read Laptops" ON public.laptops FOR SELECT USING (true);

-- Kebijakan Akses Tulis (INSERT, UPDATE, DELETE) untuk Input Manual Frontend
CREATE POLICY "Public Insert Laptops" ON public.laptops FOR INSERT WITH CHECK (true);
CREATE POLICY "Public Update Laptops" ON public.laptops FOR UPDATE USING (true);
CREATE POLICY "Public Delete Laptops" ON public.laptops FOR DELETE USING (true);

-- ==============================================================================
-- 4. INISIALISASI MASTER DATA KRITERIA (STRUKTUR MATEMATIS)
-- ==============================================================================
INSERT INTO public.kriteria (kode, nama, tipe, satuan, keterangan, default_rank) VALUES
('C1', 'Harga Beli', 'Cost', 'Rupiah', 'Budget harga pembelian laptop', 1),
('C2', 'Processor (CPU)', 'Benefit', 'Skor 0-100', 'Performa komputasi inti', 2),
('C3', 'Kapasitas RAM', 'Benefit', 'GB', 'Memori multitasking', 3),
('C4', 'Kapasitas SSD', 'Benefit', 'GB', 'Penyimpanan file dan sistem', 4),
('C5', 'Kartu Grafis (GPU)', 'Benefit', 'Skor 0-100', 'Rendering visual & gaming', 5),
('C6', 'Daya Tahan Baterai', 'Benefit', 'Wh', 'Durasi pemakaian tanpa charger', 6),
('C7', 'Portabilitas (Berat)', 'Cost', 'Kg', 'Bobot fisik mobilitas laptop', 7),
('C8', 'Kualitas Layar', 'Benefit', 'Skor 0-100', 'Panel layar, warna, dan refresh rate', 8)
ON CONFLICT (kode) DO NOTHING;
```

  <script>
    // KREDENSIAL RESMI SUPABASE
    const SUPABASE_URL = "https://jaizwaoqbqykvyvdxjyj.supabase.co";
    const SUPABASE_ANON_KEY = "sb_publishable_l9p5lowmhr4nK9-JD7B93w_tfhrTtRB";

    function spkSystem() {
      return {
        modalInput: false,
        filterStatus: 'all', // 'all', 'ready', 'indent'
        kriteriaList: [
          { id: 1, kode: 'C1', nama: 'Harga Beli', tipe: 'Cost', rank: 1, bobot: 0, keterangan: 'Nominal harga beli' },
          { id: 2, kode: 'C2', nama: 'Processor (CPU)', tipe: 'Benefit', rank: 2, bobot: 0, keterangan: 'Skor performa komputasi' },
          { id: 3, kode: 'C3', nama: 'Kapasitas RAM', tipe: 'Benefit', rank: 3, bobot: 0, keterangan: 'Kapasitas RAM (GB)' },
          { id: 4, kode: 'C4', nama: 'Kapasitas SSD', tipe: 'Benefit', rank: 4, bobot: 0, keterangan: 'Kapasitas storage (GB)' },
          { id: 5, kode: 'C5', nama: 'Kartu Grafis (GPU)', tipe: 'Benefit', rank: 5, bobot: 0, keterangan: 'Skor performa 3D & render' },
          { id: 6, kode: 'C6', nama: 'Daya Baterai', tipe: 'Benefit', rank: 6, bobot: 0, keterangan: 'Kapasitas baterai (Wh)' },
          { id: 7, kode: 'C7', nama: 'Portabilitas (Berat)', tipe: 'Cost', rank: 7, bobot: 0, keterangan: 'Bobot fisik laptop (Kg)' },
          { id: 8, kode: 'C8', nama: 'Kualitas Layar', tipe: 'Benefit', rank: 8, bobot: 0, keterangan: 'Skor resolusi & warna' }
        ],
        laptopsData: [],
        hasilRanking: [],
        formLaptop: {
          nama: '', merek: '', status: 'ready', kategori_penggunaan: 'Umum',
          harga: null, cpu_score: null, ram_gb: null, ssd_gb: null,

        },

        async init() {
          this.hitungBobotROC();
          await this.loadDataLaptops();

        async loadDataLaptops() {

            if (error) throw error;
            this.laptopsData = data || [];
          } catch (err) {

        resetPrioritas() {

          this.hitungBobotROC();
        },

        // 3. Rumus Matematis Rank Order Centroid (ROC)
        hitungBobotROC() {
          const m = this.kriteriaList.length;
          const sorted = [...this.kriteriaList].sort((a, b) => a.rank - b.rank);

          sorted.forEach((item, index) => {
            const rankIndex = index + 1;
            let harmonicSum = 0;
            for (let k = rankIndex; k <= m; k++) {
              harmonicSum += (1 / k);
            }
            const weight = harmonicSum / m;
            const target = this.kriteriaList.find(k => k.id === item.id);
            if (target) target.bobot = weight;
          });
        },

        // 4. Komputasi Algoritma TOPSIS
        kalkulasiTOPSIS() {
          this.hitungBobotROC();
          
          let dataset = [...this.laptopsData];
          if (this.filterStatus !== 'all') {
            dataset = dataset.filter(l => l.status === this.filterStatus);
          }

          if (dataset.length === 0) {
            alert("Tidak ada data laptop pada kategori filter ini. Silakan input laptop terlebih dahulu.");
            return;
          }

          const bobot = {};
          this.kriteriaList.forEach(k => bobot[k.kode] = k.bobot);

          // A. Pembagi Kuadrat Euclidean
          const pembagi = {
            c1: Math.sqrt(dataset.reduce((s, d) => s + Math.pow(Number(d.harga), 2), 0)),
            c2: Math.sqrt(dataset.reduce((s, d) => s + Math.pow(Number(d.cpu_score), 2), 0)),
            c3: Math.sqrt(dataset.reduce((s, d) => s + Math.pow(Number(d.ram_gb), 2), 0)),
            c4: Math.sqrt(dataset.reduce((s, d) => s + Math.pow(Number(d.ssd_gb), 2), 0)),
            c5: Math.sqrt(dataset.reduce((s, d) => s + Math.pow(Number(d.gpu_score), 2), 0)),
            c6: Math.sqrt(dataset.reduce((s, d) => s + Math.pow(Number(d.baterai_wh), 2), 0)),
            c7: Math.sqrt(dataset.reduce((s, d) => s + Math.pow(Number(d.berat_kg), 2), 0)),
            c8: Math.sqrt(dataset.reduce((s, d) => s + Math.pow(Number(d.layar_score), 2), 0)),
          };

          // B. Normalisasi Terbobot (Y)
          const matrixY = dataset.map(d => ({
            id: d.id,
            y1: (Number(d.harga) / (pembagi.c1 || 1)) * bobot.C1,
            y2: (Number(d.cpu_score) / (pembagi.c2 || 1)) * bobot.C2,
            y3: (Number(d.ram_gb) / (pembagi.c3 || 1)) * bobot.C3,
            y4: (Number(d.ssd_gb) / (pembagi.c4 || 1)) * bobot.C4,
            y5: (Number(d.gpu_score) / (pembagi.c5 || 1)) * bobot.C5,
            y6: (Number(d.baterai_wh) / (pembagi.c6 || 1)) * bobot.C6,
            y7: (Number(d.berat_kg) / (pembagi.c7 || 1)) * bobot.C7,
            y8: (Number(d.layar_score) / (pembagi.c8 || 1)) * bobot.C8,
          }));

          // C. Solusi Ideal Positif (A+) dan Solusi Ideal Negatif (A-)
          // Cost = Min ideal positif | Benefit = Max ideal positif
          const A_Plus = {
            y1: Math.min(...matrixY.map(m => m.y1)),
            y2: Math.max(...matrixY.map(m => m.y2)),
            y3: Math.max(...matrixY.map(m => m.y3)),
            y4: Math.max(...matrixY.map(m => m.y4)),
            y5: Math.max(...matrixY.map(m => m.y5)),
            y6: Math.max(...matrixY.map(m => m.y6)),
            y7: Math.min(...matrixY.map(m => m.y7)),
            y8: Math.max(...matrixY.map(m => m.y8))
          };

          const A_Minus = {
            y1: Math.max(...matrixY.map(m => m.y1)),
            y2: Math.min(...matrixY.map(m => m.y2)),
            y3: Math.min(...matrixY.map(m => m.y3)),
            y4: Math.min(...matrixY.map(m => m.y4)),
            y5: Math.min(...matrixY.map(m => m.y5)),
            y6: Math.min(...matrixY.map(m => m.y6)),
            y7: Math.max(...matrixY.map(m => m.y7)),
            y8: Math.min(...matrixY.map(m => m.y8))
          };

          // D. Jarak Euclidean (D+, D-) dan Nilai Preferensi (Vi)
          const hasil = dataset.map(d => {
            const y = matrixY.find(m => m.id === d.id);
            
            const dPlus = Math.sqrt(
              Math.pow(y.y1 - A_Plus.y1, 2) + Math.pow(y.y2 - A_Plus.y2, 2) +
              Math.pow(y.y3 - A_Plus.y3, 2) + Math.pow(y.y4 - A_Plus.y4, 2) +
              Math.pow(y.y5 - A_Plus.y5, 2) + Math.pow(y.y6 - A_Plus.y6, 2) +
              Math.pow(y.y7 - A_Plus.y7, 2) + Math.pow(y.y8 - A_Plus.y8, 2)
            );

            const dMinus = Math.sqrt(
              Math.pow(y.y1 - A_Minus.y1, 2) + Math.pow(y.y2 - A_Minus.y2, 2) +
              Math.pow(y.y3 - A_Minus.y3, 2) + Math.pow(y.y4 - A_Minus.y4, 2) +
              Math.pow(y.y5 - A_Minus.y5, 2) + Math.pow(y.y6 - A_Minus.y6, 2) +
              Math.pow(y.y7 - A_Minus.y7, 2) + Math.pow(y.y8 - A_Minus.y8, 2)
            );

            const skorVi = (dPlus + dMinus) === 0 ? 0 : (dMinus / (dPlus + dMinus));

            return {
              ...d,
              dPlus: dPlus,
              dMinus: dMinus,
              skorVi: skorVi
            };
          });

          // Urutkan alternatif berdasarkan nilai Vi tertinggi
          this.hasilRanking = hasil.sort((a, b) => b.skorVi - a.skorVi);

          setTimeout(() => {
            document.getElementById('hasilSection').scrollIntoView({ behavior: 'smooth' });
          }, 100);
        }
      };
